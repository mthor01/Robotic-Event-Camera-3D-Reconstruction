# Following the tutorial on https://zhuyifengzju.github.io/deoxys_docs/html/installation/codebase_installation.html
docker run -it -v /cshome:/cshome -e DISPLAY=$DISPLAY -v /tmp/.X11-unix:/tmp/.X11-unix --net=host --privileged ros:noetic

apt update && apt upgrade -y
apt install git nano wget  -y 

# add to /etc/security/limits.conf
nano /etc/security/limits.conf

@realtime soft rtprio 99
@realtime soft priority 99
@realtime soft memlock 102400
@realtime hard rtprio 99
@realtime hard priority 99
@realtime hard memlock 102400



# Install Deoxys
cd ~
git clone https://github.com/UT-Austin-RPL/deoxys_control
cd deoxys_control/deoxys
bash ./InstallPackage
# -> select 0.8.0

# export LD_LIBRARY_PATH=$(dirname "${0}")/../lib:$LD_LIBRARY_PATH fix for symlinks, but not required
make -j build_franka=1 
make -j build_deoxys=1

# for some reason the libfiles were generated into deoxys_control/deoxys, not sure why, but by moving them to ./lib it worked (nor sure if the LD_LIBRARY_PATH export above was required for this)
# MOVE Libraries to lib folder!

# Install python + conda (accept + yes, yes) (order of Deoxys install and conda environment creation is important, otherwise the protobuf from the conda environment overshadows the one installed on the system)
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh 
bash Miniconda3-latest-Linux-x86_64.sh

# restart shell / container
# build base environment (deoxys requires python <= 3.10)
conda create -n "deoxys" python=3.10 ipython ipykernel pylint jupyter pip -y

# This command definitely worked
conda create -n "deoxys" python=3.8 ipython ipykernel pylint jupyter pip pyzmq PyYAML easydict google-api-python-client pybullet numpy hidapi glfw numba termcolor simplejson mujoco==3.1.0 protobuf==3.13.0 matplotlib -c conda-forge
